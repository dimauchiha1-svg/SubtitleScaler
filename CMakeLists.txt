cmake_minimum_required(VERSION 3.21)

project(SubtitleScaler
    VERSION 1.0.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Static CRT: CommonLibSSE NG adds /MT itself, so everything else
# (vcpkg triplet x64-windows-static) is built with the static CRT as well.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(SUBTITLESCALER_DUMP "Log the menu structure when a subtitle element is not found" OFF)
option(SUBTITLESCALER_MENU "In-game settings menu for SKSE Menu Framework" ON)
set(MENU_API_DIR "" CACHE PATH "Folder with a clone of https://github.com/QTR-Modding/SKSE-Menu-Framework-3-API")

find_package(CommonLibSSE CONFIG REQUIRED)

# add_commonlibsse_plugin creates the DLL, links CommonLibSSE and generates
# SKSEPlugin_Version / SKSEPlugin_Query (Address Library, works on SE and AE).
add_commonlibsse_plugin(${PROJECT_NAME}
    AUTHOR "SubtitleScaler"
    SOURCES
        src/main.cpp
        src/Papyrus.cpp
        src/Papyrus.h
        src/Settings.cpp
        src/Settings.h
        src/Subtitles.cpp
        src/Subtitles.h
        src/UI.cpp
        src/UI.h
)

target_include_directories(${PROJECT_NAME} PRIVATE src)

# The PCH is applied to every file of the target, including the plugin file generated by CommonLibSSE.
target_precompile_headers(${PROJECT_NAME} PRIVATE src/PCH.h)

if(SUBTITLESCALER_DUMP)
    target_compile_definitions(${PROJECT_NAME} PRIVATE SUBTITLESCALER_DUMP)
endif()

# ---------------------------------------------------------------------------
# SKSE Menu Framework 3 (optional). Only the SKSEMenuFramework.h header from
# https://github.com/QTR-Modding/SKSE-Menu-Framework-3-API is needed (nothing to link: the framework
# and ImGui functions are called through GetProcAddress). If the header is not found,
# the plugin is built without the menu (a warning is printed in the CMake log).
# ---------------------------------------------------------------------------
if(SUBTITLESCALER_MENU)
    if(MENU_API_DIR AND EXISTS "${MENU_API_DIR}/SKSEMenuFramework.h")
        target_include_directories(${PROJECT_NAME} SYSTEM PRIVATE "${MENU_API_DIR}")
        target_compile_definitions(${PROJECT_NAME} PRIVATE SUBTITLESCALER_MENU)
        message(STATUS "SKSE Menu Framework: header found (${MENU_API_DIR})")
    else()
        message(WARNING "SKSE Menu Framework: SKSEMenuFramework.h not found in MENU_API_DIR='${MENU_API_DIR}'. "
                        "The plugin will be built WITHOUT the menu.")
    endif()
endif()

if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /utf-8 /permissive- /Zc:preprocessor)
endif()
