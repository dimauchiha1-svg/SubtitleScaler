cmake_minimum_required(VERSION 3.21)

project(SubtitleScaler
    VERSION 1.0.1
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Static CRT: CommonLibSSE NG adds /MT itself, so everything else
# (vcpkg triplet x64-windows-static) is built with the static CRT as well.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(SUBTITLESCALER_DUMP "Log the menu structure when a subtitle element is not found" OFF)
option(SUBTITLESCALER_MENU "In-game settings menu for SKSE Menu Framework" ON)
set(MENU_API_DIR "" CACHE PATH "Folder with a clone of https://github.com/QTR-Modding/SKSE-Menu-Framework-3-API")

# CommonLibSSE NG, alandtse fork (https://github.com/alandtse/CommonLibSSE-NG, branch "ng"). The older
# colorglass / CharmedBaryon port cannot read the Address Library V5 files of Skyrim 1.7.x, so a plugin
# built on it does not load there. This fork reads the old formats (SE 1.5.97, AE 1.6.x) and the V5 one
# (1.7.x), so one DLL works on all of them.
# The workflow clones it into lib/commonlibsse-ng; for a local build do the same (see the README).
set(COMMONLIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/lib/commonlibsse-ng" CACHE PATH "Folder with a clone of CommonLibSSE-NG (alandtse, branch ng)")
if(NOT EXISTS "${COMMONLIB_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "CommonLibSSE-NG was not found in '${COMMONLIB_DIR}'. Clone https://github.com/alandtse/CommonLibSSE-NG there (git clone --recurse-submodules --branch v16.7.0).")
endif()
set(ENABLE_SKYRIM_SE ON  CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON  CACHE BOOL "" FORCE)   # AE 1.6.x and 1.7.x
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)   # VR is not supported by this mod
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "" FORCE)   # needs a downloaded helper; the mod patches no code
set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "" FORCE)        # faster build, same result
add_subdirectory("${COMMONLIB_DIR}" commonlibsse-ng)

# With add_subdirectory the library does not define add_commonlibsse_plugin() itself (it is only loaded through
# find_package), so include the file that holds it.
include("${COMMONLIB_DIR}/cmake/CommonLibSSE.cmake")

# add_commonlibsse_plugin creates the DLL, links CommonLibSSE and generates
# the plugin declaration (SKSEPlugin_Version / SKSEPlugin_Query) with Address Library support.
add_commonlibsse_plugin(${PROJECT_NAME}
    AUTHOR "SubtitleScaler"
    SOURCES
        src/main.cpp
        src/Papyrus.cpp
        src/Papyrus.h
        src/Settings.cpp
        src/Settings.h
        src/Subtitles.cpp
        src/Subtitles.h
        src/UI.cpp
        src/UI.h
)

target_include_directories(${PROJECT_NAME} PRIVATE src)

# Windows.h defines min/max macros that break std::min / std::max. The older CommonLibSSE NG switched them off
# for its users; the alandtse fork does not, so do it here.
target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX)

# The PCH is applied to every file of the target, including the plugin file generated by CommonLibSSE.
target_precompile_headers(${PROJECT_NAME} PRIVATE src/PCH.h)

if(SUBTITLESCALER_DUMP)
    target_compile_definitions(${PROJECT_NAME} PRIVATE SUBTITLESCALER_DUMP)
endif()

# ---------------------------------------------------------------------------
# SKSE Menu Framework 3 (optional). Only the SKSEMenuFramework.h header from
# https://github.com/QTR-Modding/SKSE-Menu-Framework-3-API is needed (nothing to link: the framework
# and ImGui functions are called through GetProcAddress). If the header is not found,
# the plugin is built without the menu (a warning is printed in the CMake log).
# ---------------------------------------------------------------------------
if(SUBTITLESCALER_MENU)
    if(MENU_API_DIR AND EXISTS "${MENU_API_DIR}/SKSEMenuFramework.h")
        target_include_directories(${PROJECT_NAME} SYSTEM PRIVATE "${MENU_API_DIR}")
        target_compile_definitions(${PROJECT_NAME} PRIVATE SUBTITLESCALER_MENU)
        message(STATUS "SKSE Menu Framework: header found (${MENU_API_DIR})")
    else()
        message(WARNING "SKSE Menu Framework: SKSEMenuFramework.h not found in MENU_API_DIR='${MENU_API_DIR}'. "
                        "The plugin will be built WITHOUT the menu.")
    endif()
endif()

if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE /utf-8 /permissive- /Zc:preprocessor)
endif()
